cmake_minimum_required(VERSION 3.24)
project(ubar_webkit_port LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

if(APPLE)
  enable_language(OBJCXX)
  if(NOT UBAR_WEBKIT_FRAMEWORK_DIR)
    message(FATAL_ERROR "Set UBAR_WEBKIT_FRAMEWORK_DIR to the pinned fork's built framework directory")
  endif()
  find_library(WEBKIT_FRAMEWORK WebKit PATHS ${UBAR_WEBKIT_FRAMEWORK_DIR} NO_DEFAULT_PATH REQUIRED)
  add_library(ubar_webkit_port SHARED macos/port.mm)
  target_compile_options(ubar_webkit_port PRIVATE -fobjc-arc)
  target_link_libraries(ubar_webkit_port PRIVATE ${WEBKIT_FRAMEWORK} "-framework AppKit")
elseif(UNIX)
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(GTK3 REQUIRED IMPORTED_TARGET gtk+-3.0)
  pkg_check_modules(WEBKIT REQUIRED IMPORTED_TARGET webkit2gtk-4.1)
  add_library(ubar_webkit_port SHARED linux/port.cpp)
  target_link_libraries(ubar_webkit_port PRIVATE PkgConfig::GTK3 PkgConfig::WEBKIT)
elseif(WIN32)
  if(NOT UBAR_WEBKIT_BUILD_DIR)
    message(FATAL_ERROR "Set UBAR_WEBKIT_BUILD_DIR to the pinned fork's WebKitBuild/<configuration> directory")
  endif()
  find_library(WEBKIT_LIBRARY WebKit PATHS
    ${UBAR_WEBKIT_BUILD_DIR}/lib ${UBAR_WEBKIT_BUILD_DIR}/bin NO_DEFAULT_PATH REQUIRED)
  add_library(ubar_webkit_port SHARED windows/port.cpp)
  target_include_directories(ubar_webkit_port PRIVATE ${UBAR_WEBKIT_BUILD_DIR}/include)
  target_link_libraries(ubar_webkit_port PRIVATE ${WEBKIT_LIBRARY} user32 shell32 ole32)
else()
  message(FATAL_ERROR "Select a supported WebKit port toolchain")
endif()

target_include_directories(ubar_webkit_port PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/../../include)
set_target_properties(ubar_webkit_port PROPERTIES CXX_VISIBILITY_PRESET hidden)
